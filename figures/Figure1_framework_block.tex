%% ===========================================================================
%%  LEFT COLUMN -- the procedure

\node[pstep] (s1) at (\BX,0) {%
  \textbf{Count the baseline errors.}\\[0.7mm]
  Matched lesions $S$, reference lesions $N_{\mathrm{gt}}$, false-positive
  components $N_{\mathrm{fp}}$, missed lesions $N_{\mathrm{fn}}$, per region.};

\node[pstep,below=4.5mm of s1.south west,anchor=north west] (s2) {%
  \textbf{Bound what any method of that class could achieve.}\\[1.1mm]
  $\displaystyle
   H_{\mathrm{FP}}=\frac{S}{N_{\mathrm{gt}}}-\frac{S}{N_{\mathrm{gt}}+N_{\mathrm{fp}}}
   \qquad
   H_{\mathrm{FN}}=\frac{N_{\mathrm{fn}}}{N_{\mathrm{gt}}+N_{\mathrm{fp}}}$
  \\[1.1mm]
  Equations~\eqref{eq:fp_ceiling} and~\eqref{eq:fn_ceiling}. Both follow from the counts alone, so both are
  available before a method exists.};

\node[pstep,below=4.5mm of s2.south west,anchor=north west] (s3) {%
  \textbf{Divide the bound by the noise floor.}\\[0.7mm]
  Compare $H$ with the minimum detectable effect the design can resolve. A
  ceiling below the MDE cannot be demonstrated whatever the method.};

\node[gate,below=3.4mm of s3.south west,anchor=north west,xshift=14mm] (g3) {%
  \textbf{\textit{H} < MDE}\,---\,stop. No intervention of this class can
  produce a detectable change on this sample.};

\node[pstep,below=4.5mm of g3.south west,anchor=north west,xshift=-14mm] (s4) {%
  \textbf{Test whether the counted errors are available to act on.}\\[0.7mm]
  Sweep the proposal threshold and measure what fraction of them surfaces at
  all. A bound counts errors; it does not ask whether the model emits anything
  a rule could reach.};

\node[gate,below=3.4mm of s4.south west,anchor=north west,xshift=14mm] (g4) {%
  \textbf{Not available}\,---\,the bound is nominal. Intervene earlier, at
  training or architecture, not after the fact.};

\node[pstep,below=4.5mm of g4.south west,anchor=north west,xshift=-14mm] (s5) {%
  \textbf{Build the rule, and report the evaluable population per arm.}\\[0.7mm]
  An intervention that can empty a prediction changes $E_A\cap E_B$,
  Equation~\eqref{eq:eligible}, so a nominally paired comparison need not rest on one
  population.};

\node[note,anchor=west] (m2) at (\RX,0 |- c2) {%
  In the tumour regions, perfect false-positive removal buys
  \textbf{0.025--0.040} lesion-wise Dice; perfect recovery of every missed
  lesion buys \textbf{0.20--0.21}.};

\node[note,anchor=west] (m3) at (\RX,0 |- c3) {%
  MDE $=0.019$ Dice at $n=155$.\\[0.8mm]

  False-negative side: \textbf{10--11}\xx\ --- wide.\\[0.8mm]
  {\color{rulegrey}Tumour regions; the resection cavity differs.}};

\node[note,anchor=west] (m4) at (\RX,0 |- c4) {%
  A fifteen-fold lower threshold surfaced \textbf{21.8\%} of the missed
  lesions. \textbf{76--79\%} yield no candidate at any threshold tested, and
  the median probability inside a missed lesion is zero.};

\node[note,anchor=west] (m5) at (\RX,0 |- c5) {%
  \textbf{20 of 24} rule--region comparisons changed their evaluable set;
